🔒 رابط الداتا بيز مثبت داخلياً في الكود: يعمل تلقائياً على كل الأجهزة الجديدة بمجرد فتح الموقع بدون طلب الرابط.

إعدادات اتصال قاعدة البيانات السحابية (Google Sheets) الخاصة بالنظام:

خطوات الإعداد لأول مرة (دقيقتان فقط):
  1. افتح جدول Google Sheet جديد عبر sheets.new ثم اختر من القائمة: Extensions (الإضافات) → Apps Script.
  2. اضغط زر (📋 نسخ كود Google Apps Script الجاهز) أعلاه (أو انسخه من ملف google-apps-script/Code.gs) والصقه بالكامل ثم احفظ.
  3. اضغط Deploy (نشر) → New deployment (نشر جديد) واختر النوع Web app.
  4. اجعل Execute as: Me، واجعل Who has access: Anyone (أي شخص) ثم اضغط Deploy.
  5. انسخ الرابط الناتج (ينتهي بـ /exec) والصقه في الحقل أعلاه ثم اضغط حفظ وتفعيل المزامنة.
عرض كود Google Apps Script المدمج 
const USERS_SHEET = "_AppUsers";
const STORE_SHEET = "_AppStore";
const CHUNK_SIZE = 42000;

function doGet() {
  return jsonResponse({ ok: true, service: "Financial Attendance & Shifts Cloud Sync" });
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  try { lock.waitLock(20000); } catch (_) { return jsonResponse({ ok: false, error: "الخادم مشغول حالياً" }); }
  try {
    const payload = JSON.parse((e && e.postData && e.postData.contents) || "{}");
    const action = String(payload.action || "").trim();
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const usersSheet = ensureSheet(ss, USERS_SHEET, ["username", "displayName", "passwordHash", "token", "createdAt", "lastLoginAt"]);
    const storeSheet = ensureSheet(ss, STORE_SHEET, ["username", "revision", "updatedAt", "updatedByDevice", "dataChunk1"]);
    if (action === "ping") return jsonResponse({ ok: true, message: "تم الاتصال بقاعدة بيانات Google Sheets بنجاح" });
    if (action === "register") return handleRegister(usersSheet, storeSheet, payload);
    if (action === "login") return handleLogin(usersSheet, storeSheet, payload);
    if (action === "poll") return handlePoll(usersSheet, storeSheet, payload);
    if (action === "save") return handleSave(ss, usersSheet, storeSheet, payload);
    return jsonResponse({ ok: false, error: "عملية غير معروفة" });
  } catch (err) {
    return jsonResponse({ ok: false, error: String(err.message || err) });
  } finally {
    try { lock.releaseLock(); } catch (_) {}
  }
}

function ensureSheet(ss, name, headers) {
  let sh = ss.getSheetByName(name);
  if (!sh) { sh = ss.insertSheet(name); sh.appendRow(headers); }
  return sh;
}
function normalizeUsername(u) { return String(u || "").trim().toLowerCase(); }
function serverHash(u, h) {
  const d = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, "FIN_SYS_V1::" + normalizeUsername(u) + "::" + h, Utilities.Charset.UTF_8);
  return d.map(function(b){ const v = b < 0 ? b + 256 : b; return ("0" + v.toString(16)).slice(-2); }).join("");
}
function generateToken(u) { return serverHash(u, new Date().getTime() + "::" + Math.random()); }
function findRowByUsername(sh, u) {
  const last = sh.getLastRow(); if (last < 2) return -1;
  const vals = sh.getRange(2, 1, last - 1, 1).getValues(); const t = normalizeUsername(u);
  for (let i = 0; i < vals.length; i++) if (normalizeUsername(vals[i][0]) === t) return i + 2;
  return -1;
}
function readUserStore(sh, u) {
  const r = findRowByUsername(sh, u); if (r === -1) return null;
  const row = sh.getRange(r, 1, 1, Math.max(5, sh.getLastColumn())).getValues()[0];
  const str = row.slice(4).filter(function(c){ return c !== "" && c !== null; }).join("");
  return { revision: Number(row[1] || 0), updatedAt: String(row[2] || ""), updatedByDevice: String(row[3] || ""), workbook: str ? JSON.parse(str) : null };
}
function writeUserStore(sh, u, rev, at, dev, wb) {
  const s = JSON.stringify(wb || {}); const chunks = [];
  for (let i = 0; i < s.length; i += CHUNK_SIZE) chunks.push(s.slice(i, i + CHUNK_SIZE));
  if (!chunks.length) chunks.push("{}");
  const rowData = [normalizeUsername(u), rev, at, dev].concat(chunks);
  const r = findRowByUsername(sh, u);
  if (r === -1) sh.getRange(sh.getLastRow() + 1, 1, 1, rowData.length).setValues([rowData]);
  else { sh.getRange(r, 1, 1, Math.max(rowData.length, sh.getLastColumn())).clearContent(); sh.getRange(r, 1, 1, rowData.length).setValues([rowData]); }
}
function handleRegister(uSh, sSh, p) {
  const u = normalizeUsername(p.username);
  if (u.length < 3) return jsonResponse({ ok: false, error: "اسم المستخدم قصير" });
  if (findRowByUsername(uSh, u) !== -1) return jsonResponse({ ok: false, error: "اسم المستخدم مسجل بالفعل" });
  const now = new Date().toISOString(); const token = generateToken(u);
  uSh.appendRow([u, p.displayName || u, serverHash(u, p.passwordHash), token, now, now]);
  if (p.initialWorkbook) writeUserStore(sSh, u, 1, now, p.deviceId || "Web", p.initialWorkbook);
  return jsonResponse({ ok: true, user: { username: u, displayName: p.displayName || u, token: token }, revision: 1, workbook: p.initialWorkbook || null });
}
function handleLogin(uSh, sSh, p) {
  const u = normalizeUsername(p.username); const r = findRowByUsername(uSh, u);
  if (r === -1) return jsonResponse({ ok: false, error: "اسم المستخدم غير موجود" });
  const row = uSh.getRange(r, 1, 1, 6).getValues()[0];
  if (String(row[2]) !== serverHash(u, p.passwordHash)) return jsonResponse({ ok: false, error: "كلمة المرور غير صحيحة" });
  let token = String(row[3] || ""); if (!token) { token = generateToken(u); uSh.getRange(r, 4).setValue(token); }
  uSh.getRange(r, 6).setValue(new Date().toISOString());
  const st = readUserStore(sSh, u);
  return jsonResponse({ ok: true, user: { username: u, displayName: String(row[1] || u), token: token }, revision: st ? st.revision : 0, updatedByDevice: st ? st.updatedByDevice : "", workbook: st ? st.workbook : null });
}
function verifyToken(uSh, u, t) { const r = findRowByUsername(uSh, u); return r !== -1 && String(uSh.getRange(r, 4).getValue()) === String(t); }
function handlePoll(uSh, sSh, p) {
  const u = normalizeUsername(p.username);
  if (!verifyToken(uSh, u, p.token)) return jsonResponse({ ok: false, error: "انتهت الجلسة" });
  const st = readUserStore(sSh, u);
  if (!st || st.revision <= Number(p.clientRevision || 0)) return jsonResponse({ ok: true, hasUpdate: false, revision: st ? st.revision : 0 });
  return jsonResponse({ ok: true, hasUpdate: true, revision: st.revision, updatedByDevice: st.updatedByDevice, workbook: st.workbook });
}
function handleSave(ss, uSh, sSh, p) {
  const u = normalizeUsername(p.username);
  if (!verifyToken(uSh, u, p.token)) return jsonResponse({ ok: false, error: "انتهت الجلسة" });
  const st = readUserStore(sSh, u); const nextRev = (st ? st.revision : 0) + 1; const now = new Date().toISOString();
  writeUserStore(sSh, u, nextRev, now, p.deviceId || "Web", p.workbook);
  if (p.readableSummary && p.readableSummary.rows && p.readableSummary.rows.length) {
    try {
      let m = ss.getSheetByName(("ملخص_" + u).slice(0, 28)) || ss.insertSheet(("ملخص_" + u).slice(0, 28));
      m.clearContents(); m.getRange(1, 1, p.readableSummary.rows.length, p.readableSummary.rows[0].length).setValues(p.readableSummary.rows);
    } catch (_) {}
  }
  return jsonResponse({ ok: true, revision: nextRev, updatedAt: now });
}
function jsonResponse(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
